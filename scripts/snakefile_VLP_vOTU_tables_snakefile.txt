configfile: "config_VLP_vOTU_tables.yaml"
f1 = open(config["sample_file"],"r")
SAMPLES = f1.read().splitlines()
f1.close()
rule all:
	input:
		expand([config["vOTU_tables_dir"]+"/{sample}_wQ33_vOTUs_coverage_nodecontam.tsv"],sample=SAMPLES)
rule bowtie2_align:
	input:
		read1=config["reads_dir"]+"/{sample}_trimmed_clean_paired_1.fastq.gz",
		read2=config["reads_dir"]+"/{sample}_trimmed_clean_paired_2.fastq.gz"
	params: 
		tmp_cat=config["vlp_mapping_directory"]+"/{sample}/{sample}_trimmed_clean_allreads.fastq",
		r_d=config["reads_dir"],
		bulkmapdir=config["bulk_mapping_directory"],
		vlpmapdir=config["vlp_mapping_directory"],
		NGS="{sample}"
	singularity:
		config["extendApptainer"]
	output: 
		config["vlp_mapping_directory"]+"/{sample}/{sample}_vOTUs.sorted.bam"
	threads: 20
	shell:
		"""
		mkdir {params.vlpmapdir}/{params.NGS}
		zcat {input.read1} {input.read2} > {params.tmp_cat}
		bowtie2 -p {threads} -x {params.bulkmapdir}/cleanedrep_seqs_vOTUs_noIMGVR_renamed -U {params.tmp_cat} -S {params.vlpmapdir}/{params.NGS}/{params.NGS}_vOTUs.sam
		samtools sort -o {output} {params.vlpmapdir}/{params.NGS}/{params.NGS}_vOTUs.sam && rm {params.vlpmapdir}/{params.NGS}/{params.NGS}_vOTUs.sam
		rm {params.tmp_cat}
		"""
rule coverm:
	input:
		bam=config["vlp_mapping_directory"]+"/{sample}/{sample}_vOTUs.sorted.bam"
	params:
		outdir=config["vOTU_tables_dir"],
		samp=temp("{sample}")
	singularity:
		config["extendApptainer"]
	output:
		config["vOTU_tables_dir"]+"/{sample}_wQ33_vOTUs_coverage_nodecontam.tsv"
	threads: 20
	resources:
		mem_mb=4000*4,
		disk_mb=10000
	shell:
		"""
		coverm contig -t 20 --min-covered-fraction 75 --methods tpm --bam-files {input.bam} > {params.outdir}/{params.samp}_wQ33_vOTUs_coverage_nodecontam.tsv
		"""
