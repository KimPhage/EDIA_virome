SAMPLES = f1.read().splitlines()

	input:
		expand([config["CheckVOut_dir"]+"/{sample}/{sample}_Proviruses_CheckV_Output/complete_genomes.tsv",
		config["CheckVOut_dir"]+"/{sample}/{sample}_Proviruses_CheckV_Output/completeness.tsv",

		config["CheckVOut_dir"]+"/{sample}/{sample}_Proviruses_CheckV_Output/viruses.fna"],sample=SAMPLES)

rule geNomad_run:
	input:
		contigs=config["CheckVOut_dir"]+"/{sample}/{sample}_Viruses_CheckV_Output/proviruses.fna"
	params:
		gNOutDir=config["gNOutDir_path"], #remove this if script does not run
		gNdb=config["gNdb_path"],
		NGS=temp("{sample}")
	singularity:
		config["GNomad_apptainer"]
	output:
		config["gNOutDir_path"]+"/{sample}.checkV_proviruses/proviruses_summary/proviruses_virus.fna" 
	threads: 20
	resources:
		mem_mb=50000
	shell:
		"""
		genomad end-to-end -t {threads} {input.contigs} {params.gNOutDir}/{params.NGS}.checkV_proviruses {params.gNdb}
		"""
