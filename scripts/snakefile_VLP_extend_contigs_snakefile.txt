configfile: "config_extend.yaml"
f1 = open(config["sample_file"],"r")
SAMPLES = f1.read().splitlines()
f1.close()
rule all:
	input:
		expand([config["assemby_dir"]+"/COBRA/{sample}_clean_VLP_contigs_new.fa"],sample=SAMPLES)
rule assembly:
	input:
		pair1=config["clean_reads"]+"/{sample}_trimmed_clean_noQ33_controlrm_paired_1.fastq.gz",
		pair2=config["clean_reads"]+"/{sample}_trimmed_clean_noQ33_controlrm_paired_2.fastq.gz",
		unmatched=config["clean_reads"]+"/{sample}_trimmed_clean_noQ33_controlrm_unmatched.fastq.gz"
	params:
		outdir=config["assemby_dir"],
		samp=temp("{sample}")
	singularity:
		config["extendApptainer"]
	output:
		config["assemby_dir"]+"/{sample}_clean_VLP_contigs.fa"
	threads: 20
	resources:
		mem_mb=4000*4,
		disk_mb=10000
	shell:
		"""
		megahit -1 {input.pair1} -2 {input.pair2} -r {input.unmatched} -o {params.outdir}/{params.samp}
		mv {params.outdir}/{params.samp}/final.contigs.fa {params.outdir}/{params.samp}_clean_VLP_contigs.fa
		"""
rule indexing_mapping:
	input:
		contigs=config["assemby_dir"]+"/{sample}_clean_VLP_contigs.fa",
		pair1=config["clean_reads"]+"/{sample}_trimmed_clean_noQ33_controlrm_paired_1.fastq.gz",
		pair2=config["clean_reads"]+"/{sample}_trimmed_clean_noQ33_controlrm_paired_2.fastq.gz",
		unmatched=config["clean_reads"]+"/{sample}_trimmed_clean_noQ33_controlrm_unmatched.fastq.gz"
	params:
		outdir=config["assemby_dir"],
		samp=temp("{sample}")
	singularity:
		config["extendApptainer"]
	output:
		config["assemby_dir"]+"/{sample}_clean_VLP_assembly_aligned_sorted.bam"
	threads: 20
	resources:
		mem_mb=4000*4,
		disk_mb=10000
	shell:
		"""
		bowtie2-build {input.contigs} {params.outdir}/{params.samp}_clean_VLP_contigs
		bowtie2 -p {threads} -x {params.outdir}/{params.samp}_clean_VLP_contigs -U {input.pair1},{input.pair2},{input.unmatched} -S {params.outdir}/{params.samp}_clean_VLP_assembly_aligned.sam
		samtools sort -@ {threads} {params.outdir}/{params.samp}_clean_VLP_assembly_aligned.sam -o {params.outdir}/{params.samp}_clean_VLP_assembly_aligned_sorted.bam
		rm {params.outdir}/{params.samp}_clean_VLP_assembly_aligned.sam
		rm {params.outdir}/*.bt2
		"""
rule calculate_coverage:
	input:
		bam=config["assemby_dir"]+"/{sample}_clean_VLP_assembly_aligned_sorted.bam"
	params:
		outdir=config["assemby_dir"],
		samp=temp("{sample}")
	singularity:
		config["extendApptainer"]
	output:
		config["assemby_dir"]+"/{sample}_clean_VLP_assembly_coverage.txt"
	threads: 20
	resources:
		mem_mb=4000*4,
		disk_mb=10000
	shell:
		"""
		coverm contig --methods metabat --bam-files {params.outdir}/{params.samp}_clean_VLP_assembly_aligned_sorted.bam > {params.outdir}/{params.samp}_clean_VLP_assembly_coverage.txt
		"""
rule Cobra
	input:
		coverage=config["assemby_dir"]+"/{sample}_clean_VLP_assembly_coverage.txt",
		bam=config["assemby_dir"]+"/{sample}_clean_VLP_assembly_aligned_sorted.bam",
		contigs=config["assemby_dir"]+"/{sample}_clean_VLP_contigs.fa"
	params:
		outdir=config["assemby_dir"],
		samp=temp("{sample}")
	singularity:
		config["extendApptainer"]
	output:
		config["assemby_dir"]+"/COBRA/{sample}_clean_VLP_contigs_new.fa"
	threads: 20
	resources:
		mem_mb=4000*4,
		disk_mb=10000
	shell:
		"""
		cut -f 1,3 {input.coverage} | sed 1d > {params.outdir}/{params.samp}_clean_VLP_assembly_coverage_trunc.txt
		cobra-meta -f {input.contigs} -q {input.contigs} -o {params.outdir}/COBRA -c {params.outdir}/{params.samp}_clean_VLP_assembly_coverage_trunc.txt -m {input.bam} -a megahit -mink 21 -maxk 141 
		"""
